calls=[]
for name in ['core/services.py','core/coordination.py']:
    tree=ast.parse((APP/name).read_text('utf-8'))
    for node in ast.walk(tree):
        if not isinstance(node,ast.Call) or len(node.args)<3:continue
        fn=node.func
        is_notify=(isinstance(fn,ast.Name) and fn.id=='notify') or (isinstance(fn,ast.Attribute) and fn.attr=='notify')
        if is_notify and isinstance(node.args[2],ast.Constant) and isinstance(node.args[2].value,str):
            kind=node.args[2].value
            if kind not in LABELS:calls.append(dict(kind=kind,location=name+':'+str(node.lineno),label=business_label(kind)))
(OUT/'通知类型字典核对.json').write_text(json.dumps(calls,ensure_ascii=False,indent=2)+'\n','utf-8')

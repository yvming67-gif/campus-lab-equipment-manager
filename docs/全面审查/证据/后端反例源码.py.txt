def note(name,result):
    ROWS.append({'name':name,**result});(HERE/'observations.json').write_text(json.dumps(ROWS,ensure_ascii=False,indent=2)+'\n','utf8')

    def setUpTestData(cls):call_command('seed_demo',verbosity=0)
    def setUp(self):

class DeliveryLeaseObservation(TransactionTestCase):
    @override_settings(DEMO_MODE=False)
    def test_observe_stale_failure_overwrites_new_lease_acceptance(self):
        call_command('seed_demo',verbosity=0)
        admin=User.objects.get(username='admin');student=User.objects.get(username='student');eq=Resource.objects.get(code='EQ-OSC-01')
        at=timezone.now();ev=s.audit(admin,eq,'AUDIT_PROBE',reason='隔离超时租约模拟')
        s.notify(ev,[student],'AUDIT_PROBE','独立模拟通知',at)
        task=Delivery.objects.get(channel='EXTERNAL',notification__audit=ev)
        for i in range(3):DeliveryAttempt.objects.create(delivery=task,attempt_no=i,outcome='ERROR',detail='模拟已失败尝试',attempted_at=at-timedelta(minutes=30-i))
        a_enter,b_enter,a_release,b_release=Event(),Event(),Event(),Event();outcomes={}
        def adapter(_):
            if current_thread().name=='old-lease':
                a_enter.set();assert a_release.wait(10);raise TimeoutError('simulated old request failure')
            b_enter.set();assert b_release.wait(10);return {'accepted':True,'receipt_id':'simulated-new-lease-success'}
        def run(name,when):
            try:outcomes[name]=s.process_delivery(task.pk,at=when)
            except Exception as e:outcomes[name]=type(e).__name__
            finally:connections.close_all()
        with patch('core.adapters.external_delivery',side_effect=adapter):
            old=Thread(target=run,name='old-lease',args=('old',at));old.start();self.assertTrue(a_enter.wait(10))
            new=Thread(target=run,name='new-lease',args=('new',at+timedelta(seconds=31)));new.start();self.assertTrue(b_enter.wait(10))
            a_release.set();old.join(10);self.assertFalse(old.is_alive())
            b_release.set();new.join(10);self.assertFalse(new.is_alive())
        task.refresh_from_db()
        note('delivery_stale_lease',{'initial_attempts':3,'two_adapter_calls':'old failure after second lease acquired, then second provider success','processing_returns':outcomes,'final_status':task.status,'success_attempt_count':task.attempts.filter(outcome='SUCCESS').count(),'last_error':task.last_error,'external_provider':'mock adapter only, no real delivery'})
        self.assertEqual(task.status,'FAILED');self.assertEqual(task.attempts.filter(outcome='SUCCESS').count(),0)
